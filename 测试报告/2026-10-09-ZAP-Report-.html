document.addEventListener('DOMContentLoaded', function () {
    const tabSms = document.getElementById('tabSms');
    const tabPwd = document.getElementById('tabPwd');
    const panelSms = document.getElementById('panelSms');
    const panelPwd = document.getElementById('panelPwd');
    const msgBox = document.getElementById('msgBox');

    const smsPhone = document.getElementById('smsPhone');
    const smsCode = document.getElementById('smsCode');
    const sendCodeBtn = document.getElementById('sendCodeBtn');
    const smsLoginBtn = document.getElementById('smsLoginBtn');

    const pwdUsername = document.getElementById('pwdUsername');
    const pwdPassword = document.getElementById('pwdPassword');
    const pwdLoginBtn = document.getElementById('pwdLoginBtn');

    let countdown = 0;
    let timer = null;

    function showMsg(text, type) {
        msgBox.textContent = text;
        msgBox.className = 'msg ' + type;
    }
    function hideMsg() { msgBox.className = 'msg'; }

    // Tab
    tabSms.addEventListener('click', function () {
        tabSms.classList.add('active'); tabPwd.classList.remove('active');
        panelSms.classList.add('active'); panelPwd.classList.remove('active');
        hideMsg();
    });
    tabPwd.addEventListener('click', function () {
        tabPwd.classList.add('active'); tabSms.classList.remove('active');
        panelPwd.classList.add('active'); panelSms.classList.remove('active');
        hideMsg();
    });

    // 手机验证码
    function checkSendBtn() {
        const phone = smsPhone.value.trim();
        const valid = /^1[3-9]\d{9}$/.test(phone);
        sendCodeBtn.disabled = !(valid && countdown === 0);
        sendCodeBtn.textContent = countdown > 0 ? (countdown + ' 秒后重发') : '发送验证码';
    }
    function checkSmsLoginBtn() {
        const vp = /^1[3-9]\d{9}$/.test(smsPhone.value.trim());
        const vc = /^\d{6}$/.test(smsCode.value.trim());
        smsLoginBtn.disabled = !(vp && vc);
    }
    smsPhone.addEventListener('input', function () { checkSendBtn(); checkSmsLoginBtn(); });
    smsCode.addEventListener('input', checkSmsLoginBtn);
    checkSendBtn();
    checkSmsLoginBtn();

    // ★ 发送验证码：后端返回 code → alert 显示 → 自动填入
    sendCodeBtn.addEventListener('click', function () {
        const phone = smsPhone.value.trim();
        if (!/^1[3-9]\d{9}$/.test(phone)) { showMsg('请输入正确的手机号', 'error'); return; }
        sendCodeBtn.disabled = true;

        fetch('/api/user/send-sms', {
            method: 'POST',
            headers: { 'Content-Type': 'application/json' },
            body: JSON.stringify({ phone: phone })
        })
        .then(r => r.json())
        .then(result => {
            if (result.code !== 200) throw new Error(result.message);
            const code = result.data;

            // ★ 弹窗显示
            alert('【演示环境】\n您的验证码是：' + code + '\n\n已自动填入。');
            smsCode.value = code;
            checkSmsLoginBtn();

            countdown = 60;
            checkSendBtn();
            timer = setInterval(function () {
                countdown--;
                checkSendBtn();
                if (countdown <= 0) { clearInterval(timer); checkSendBtn(); }
            }, 1000);
        })
        .catch(err => {
            showMsg(err.message || '发送失败', 'error');
            sendCodeBtn.disabled = false;
            checkSendBtn();
        });
    });

    // 短信登录
    smsLoginBtn.addEventListener('click', doSmsLogin);
    function doSmsLogin() {
        const phone = smsPhone.value.trim();
        const code = smsCode.value.trim();
        if (!/^1[3-9]\d{9}$/.test(phone)) { showMsg('请输入正确的手机号', 'error'); return; }
        if (!/^\d{6}$/.test(code)) { showMsg('请输入 6 位验证码', 'error'); return; }

        smsLoginBtn.disabled = true;
        smsLoginBtn.textContent = '登录中...';

        fetch('/api/user/login-sms', {
            method: 'POST',
            headers: { 'Content-Type': 'application/json' },
            body: JSON.stringify({ phone: phone, code: code })
        })
        .then(r => r.json())
        .then(result => {
            if (result.code !== 200) throw new Error(result.message);
            showMsg('登录成功，跳转中...', 'success');
            setTimeout(() => {
                const back = sessionStorage.getItem('loginBack') || '/';
                sessionStorage.removeItem('loginBack');
                window.location.href = back;
            }, 700);
        })
        .catch(err => {
            showMsg(err.message || '登录失败', 'error');
            smsLoginBtn.disabled = false;
            smsLoginBtn.textContent = '登录 / 注册';
            checkSmsLoginBtn();
        });
    }

    // 账号密码
    function checkPwdLoginBtn() {
        const u = pwdUsername.value.trim();
        const p = pwdPassword.value.trim();
        pwdLoginBtn.disabled = !(u && p);
    }
    pwdUsername.addEventListener('input', checkPwdLoginBtn);
    pwdPassword.addEventListener('input', checkPwdLoginBtn);
    checkPwdLoginBtn();

    pwdLoginBtn.addEventListener('click', function () {
        const username = pwdUsername.value.trim();
        const password = pwdPassword.value.trim();
        if (!username || !password) { showMsg('请填写用户名和密码', 'error'); return; }

        pwdLoginBtn.disabled = true;
        pwdLoginBtn.textContent = '登录中...';

        fetch('/api/user/login', {
            method: 'POST',
            headers: { 'Content-Type': 'application/json' },
            body: JSON.stringify({ username: username, password: password })
        })
        .then(r => r.json())
        .then(result => {
            if (result.code !== 200) throw new Error(result.message);
            showMsg('登录成功，跳转中...', 'success');
            setTimeout(() => {
                const back = sessionStorage.getItem('loginBack') || '/';
                sessionStorage.removeItem('loginBack');

        })
        .catch(err => {
            showMsg(err.message || '登录失败', 'error');
            pwdLoginBtn.disabled = false;
            pwdLoginBtn.textContent = '登录';
            checkPwdLoginBtn();

    });
